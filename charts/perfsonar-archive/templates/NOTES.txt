1. Get the OpenSearch Dashboards URL by running these commands:
  export NODE_PORT=$(kubectl get --namespace {{ .Release.Namespace }} -o jsonpath="{.spec.ports[0].nodePort}" services {{ include "perfsonar-archive.fullname" . }}-opensearch-dashboards)
  export NODE_IP=$(kubectl get nodes --namespace {{ .Release.Namespace }} -o jsonpath="{.items[0].status.addresses[0].address}")
  echo http://$NODE_IP:$NODE_PORT

2. To get the admin password for OpenSearch run:
   kubectl exec -it statefulset/{{ include "perfsonar-archive.fullname" . }}-opensearch-node -- grep -w admin /usr/lib/perfsonar/archive/auth_setup.out
